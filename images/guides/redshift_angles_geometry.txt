Conceptual illustration generated with matplotlib and PyAutoLens, 2026-11-17, for guides/units/redshifts_and_angles.py. The side-on sketch is not to scale (bending angle exaggerated); the two lensed images are real PyAutoLens ray-tracing of the same Isothermal lens (einstein_radius = 1.6") at (z_lens, z_source) = (0.5, 1.0) and (0.2, 2.0), and their largest pixel difference is 0.0. Distances and Einstein masses use Planck15. To regenerate, save the producer script below as a .py file and run it from the autolens_workspace root.

Producer script:

"""
Producer for images/guides/redshift_angles_geometry.png.

Run from the autolens_workspace root: python <this script>
"""

import numpy as np
import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt
import autolens as al

out = "images/guides/redshift_angles_geometry.png"

grid = al.Grid2D.uniform(shape_native=(120, 120), pixel_scales=0.05)
cosmology = al.cosmo.Planck15()
setups = [("Setup A", 0.5, 1.0, "#1f77b4"), ("Setup B", 0.2, 2.0, "#d62728")]

images, info = [], []

for name, z_lens, z_source, colour in setups:
    lens = al.Galaxy(
        redshift=z_lens,
        mass=al.mp.Isothermal(
            centre=(0.0, 0.0), ell_comps=(0.1, 0.0), einstein_radius=1.6
        ),
    )
    source = al.Galaxy(
        redshift=z_source,
        bulge=al.lp.SersicCore(
            centre=(0.1, 0.1),
            ell_comps=(0.1, 0.0),
            intensity=1.0,
            effective_radius=0.3,
            sersic_index=1.0,
        ),
    )
    tracer = al.Tracer(galaxies=[lens, source])
    images.append(np.asarray(tracer.image_2d_from(grid=grid).native))

    mass_angular = al.LensCalc.from_tracer(tracer=tracer).einstein_mass_angular_from(
        grid=grid
    )
    mass = mass_angular * cosmology.critical_surface_density_between_redshifts_from(
        redshift_0=z_lens, redshift_1=z_source
    )
    d_l = cosmology.angular_diameter_distance_to_earth_in_kpc_from(redshift=z_lens) / 1e6
    d_s = (
        cosmology.angular_diameter_distance_to_earth_in_kpc_from(redshift=z_source) / 1e6
    )
    info.append((d_l, d_s, mass))

diff = np.abs(images[0] - images[1]).max()
mass_max = max(i[2] for i in info)


def sci(value):
    exponent = int(np.floor(np.log10(value)))
    return rf"{value / 10**exponent:.1f} $\times$ 10$^{{{exponent}}}$ M$_\odot$"


def label_block(ax, x, y, lines, ha="center", va="top", fontsize=12):
    """Write one text line per entry at a fixed baseline spacing, so a line holding
    mathtext does not stretch the gaps of a multi-line label."""
    step = 1.45 * fontsize  # points between baselines
    if va == "top":
        first = -fontsize  # first baseline one cap height below the anchor
    else:
        first = 0.5 * step * (len(lines) - 1) - 0.35 * fontsize  # centre on anchor
    for i, line in enumerate(lines):
        ax.annotate(
            line,
            xy=(x, y),
            xytext=(0, first - i * step),
            textcoords="offset points",
            ha=ha,
            va="baseline",
            fontsize=fontsize,
        )


fig, axes = plt.subplots(
    2, 2, figsize=(12, 9), gridspec_kw=dict(width_ratios=[1.75, 1.0])
)

theta = 0.28  # exaggerated viewing angle for the sketch (radians)
h_max = max(d_l for d_l, _, _ in info) * np.tan(theta)  # deepest ray in either row

for row, ((name, z_lens, z_source, colour), (d_l, d_s, mass), image) in enumerate(
    zip(setups, info, images)
):
    ax = axes[row, 0]
    h_lens = d_l * np.tan(theta)

    for sign in (1, -1):
        ax.plot([0, d_l], [0, sign * h_lens], color=colour, lw=2.5)
        ax.plot([d_l, d_s], [sign * h_lens, 0], color=colour, lw=2.5, ls="--")

    ax.plot([0, d_s], [0, 0], color="0.75", lw=0.8, zorder=0)

    ax.scatter([0], [0], s=260, color="#2ca02c", zorder=5)
    label_block(ax, 0, -0.12, ["Earth"])

    # The lens label hangs at the same height in both rows, just below the deepest ray.
    ax.scatter([d_l], [0], s=150 + 900 * mass / mass_max, color="0.25", zorder=5)
    ax.plot([d_l, d_l], [-0.1, -h_max - 0.07], color="0.6", lw=0.8, zorder=1)
    label_block(
        ax,
        d_l,
        -h_max - 0.1,
        [
            "lens galaxy",
            f"z = {z_lens}",
            f"{d_l:.2f} Gpc away",
            f"mass inside 1.6″: {sci(mass)}",
        ],
    )

    ax.scatter(
        [d_s], [0], s=260, marker="*", color="orange", edgecolor="k", zorder=5
    )
    label_block(
        ax,
        d_s + 0.08,
        0.0,
        ["source galaxy", f"z = {z_source}", f"{d_s:.2f} Gpc away"],
        ha="left",
        va="center",
    )

    # The Einstein radius is the angle between the optical axis and one ray, so the
    # arc spans the upper half of the opening only, with its label inside that angle.
    arc = np.linspace(0.0, theta, 50)
    ax.plot(0.4 * np.cos(arc), 0.4 * np.sin(arc), color="k", lw=1.5)
    ax.text(
        0.44,
        0.064,
        "1.6″",
        ha="left",
        va="center",
        fontsize=12,
        fontweight="bold",
        zorder=4,
    )

    ax.set_xlim(-0.12, 2.12)
    ax.set_ylim(-1.0, 0.5)
    ax.axis("off")
    ax.set_title(
        f"{name}: lens z = {z_lens}, source z = {z_source}",
        color=colour,
        fontsize=14,
        fontweight="bold",
        loc="left",
        pad=10,
    )

    ax = axes[row, 1]
    ax.imshow(image, cmap="inferno", extent=[-3, 3, -3, 3], origin="lower")
    ax.set_title("What the telescope sees", color=colour, fontsize=12.5, pad=10)
    ax.set_xticks([-2, 0, 2])
    ax.set_yticks([-2, 0, 2])
    ax.tick_params(labelsize=10.5)
    ax.set_xlabel("x (arcsec)", fontsize=11)
    ax.set_ylabel("y (arcsec)", fontsize=11, labelpad=2)

fig.suptitle(
    "Different redshifts, same angle seen from Earth, identical image\n"
    f"(largest pixel difference between A and B = {diff:.1f})",
    fontsize=16,
    linespacing=1.4,
    y=0.975,
)
fig.text(
    0.025,
    0.025,
    "Not to scale: the bending angle is exaggerated. Distances are angular diameter distances (Planck15).\n"
    "Lens mass is the mass inside the 1.6″ Einstein radius.",
    fontsize=10,
    color="0.35",
    linespacing=1.4,
)
fig.subplots_adjust(
    left=0.025, right=0.975, top=0.85, bottom=0.13, hspace=0.38, wspace=0.12
)
fig.savefig(out, dpi=150)
print(out, diff)
